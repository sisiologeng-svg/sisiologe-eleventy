const https = require('https'); const crypto = require('crypto'); exports.handler = async (event) => { console.log('Webhook received:', event.httpMethod); if (event.httpMethod !== 'POST') { return { statusCode: 405, body: 'Method not allowed' }; } try { // Verify Paystack signature const secret = process.env.PAYSTACK_SECRET_KEY; const hash = crypto.createHmac('sha512', secret) .update(event.body) .digest('hex'); console.log('Signature check:', hash === event.headers['x-paystack-signature']); if (hash !== event.headers['x-paystack-signature']) { console.log('Invalid signature'); return { statusCode: 401, body: 'Invalid signature' }; } const payload = JSON.parse(event.body); console.log('Event type:', payload.event); if (payload.event !== 'charge.success') { return { statusCode: 200, body: 'OK' }; } const metadata = payload.data.metadata; const customFields = metadata?.custom_fields || []; const productSlug = customFields.find( f => f.variable_name === 'product_slug' )?.value; console.log('Product slug:', productSlug); if (!productSlug) { console.log('No product slug found in metadata'); return { statusCode: 200, body: 'No product slug' }; } const token = process.env.GITHUB_TOKEN; const repo = process.env.GITHUB_REPO; // Get the file from GitHub const getFile = () => new Promise((resolve, reject) => { const options = { hostname: 'api.github.com', path: `/repos/${repo}/contents/src/products/${productSlug}.md`, headers: { 'Authorization': `token ${token}`, 'User-Agent': 'sisiologe-webhook' } }; const req = https.get(options, (res) => { let data = ''; res.on('data', chunk => data += chunk); res.on('end', () => resolve(JSON.parse(data))); }); req.on('error', reject); }); const fileData = await getFile(); console.log('File found:', fileData.name); const content = Buffer.from(fileData.content, 'base64').toString('utf8'); const updated = content.replace('available: true', 'available: false'); const newContent = Buffer.from(updated).toString('base64'); // Update the file const updateFile = () => new Promise((resolve, reject) => { const body = JSON.stringify({ message: `Mark ${productSlug} as sold out`, content: newContent, sha: fileData.sha }); const options = { hostname: 'api.github.com', path: `/repos/${repo}/contents/src/products/${productSlug}.md`, method: 'PUT', headers: { 'Authorization': `token ${token}`, 'User-Agent': 'sisiologe-webhook', 'Content-Type': 'application/json', 'Content-Length': Buffer.byteLength(body) } }; const req = https.request(options, (res) => { let data = ''; res.on('data', chunk => data += chunk); res.on('end', () => resolve(JSON.parse(data))); }); req.on('error', reject); req.write(body); req.end(); }); await updateFile(); console.log('Product marked as sold out successfully!'); return { statusCode: 200, body: 'Product marked as sold out' }; } catch (error) { console.log('Error:', error.message); return { statusCode: 200, body: 'Error handled' }; } };
